// PROBE D1: With(k) then WithGroup(k) on the same logger.
func TestZZProbeGroupCollisionPanics(t *testing.T) {
	l := newDebugLog(t, Options{})
	defer func() {
		if r := recover(); r != nil {
			t.Logf("PROBE D1 panic on caller goroutine: %v", r)
		}
		_ = l.Close()
	}()
	l.Logger().With("http", "plain").WithGroup("http").Info("probe", "k", "v")
	t.Logf("PROBE D1 no panic")
}

// PROBE D4: torn last line before restart.
func TestZZProbeTornLine(t *testing.T) {
	dir := t.TempDir()
	if err := os.WriteFile(filepath.Join(dir, FileName), []byte(`{"time":"2026-01-01T00:00:00Z","level":"INFO","msg":"cut`), 0o600); err != nil {
		t.Fatal(err)
	}
	l := newDebugLog(t, Options{Dir: dir})
	l.Logger().Info("after restart")
	closeLog(t, l)
	got, err := l.Query(context.Background(), Query{MinLevel: slog.LevelDebug})
	t.Logf("PROBE D4 query after restart: n=%d err=%v", len(got), err)
}

// PROBE D8: value shapes in free strings.
func TestZZProbeValueShapes(t *testing.T) {
	for _, s := range []string{
		"https://user:p@ss@host/x",
		"-----BEGIN RSA PRIVATE KEY-----\nMIIEowIBAAKCAQEAsecretbody",
		"Authorization: Basic dXNlcjpwYXNzd29yZA==",
		"curl -H 'Authorization: Bearer abcdefghijkl'",
		fmt.Sprintf("key=%s", "sk-ant-api03-abcdefghijklmnop"),
	} {
		t.Logf("PROBE D8 %q -> %q", s, MaskSecrets(s))
	}
}
